# click_pick_place.py - 클릭 선택 기반 파지 및 배치 (Joint-Lag 충돌 방지)
import sys, time, cv2
import config, gripper, perception
import task_state_machine as tsm
from kinematics import CollisionDetected, SOArm101

WINDOW = "click the cube, then the bin - 'q' to cancel"

def run_task():
    arm = SOArm101()
    arm.connect()
    try:
        arm.move_to_home()
        click_px = wait_for_user_click(WINDOW)
        if click_px is None:
            return
        
        target_xyz = perception.pixel_to_table_xyz(click_px)
        tsm.fine_servo_and_grasp(arm, target_xyz)
    except CollisionDetected as e:
        print(f"[COLLISION DETECTED] {e} -> retreating to home")
        arm.retreat_to_last_good_pose()
    finally:
        arm.move_to_home()

if __name__ == "__main__":
    run_task()
